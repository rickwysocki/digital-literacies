\begin{frame}[label={sec:org0558820}]{Selber's Multiliteracies}
\begin{itemize}
\item \alert{Functional literacy:} Our ability to effectively \emph{use} tools and technologies for communciation tasks.
\item \alert{Critical literacy:} Our Our ability to analyze and critique the use and effects of tools and technologies at the individual, organizational, and public levels.
\item \alert{Rhetorical literacy:} Our ability to consider and employ tools and technologies for persuasive goals.
\end{itemize}
\end{frame}

\begin{frame}[label={sec:org7a4478b}]{Agency and Determinism}
\begin{itemize}
\item \alert{Agency:} The degree of freedom and control we have in a particular situation.
\item \alert{Determinism:} The degree to which a particular situation determines our possibilities.
\end{itemize}
\end{frame}
\begin{frame}[label={sec:org93d7565}]{Affordances and Limitations}
\begin{itemize}
\item \alert{Affordances:} The uses, activities, and practices that are afforded, suggested, and/or incentivized by a tool or technology. More simply: What a tool allows or suggests taht you do with it.
\item \alert{Limitations:} Uses, activities, and practices that are limited or impossible with a particular tool.
\end{itemize}
\end{frame}
