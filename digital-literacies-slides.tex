\begin{frame}[label={sec:orgd110854}]{Data Ideologies}
\begin{itemize}
\item Data is "objective"
\item Data "represents" reality
\item Data is "hard" whereas other forms of knowledge are "soft"
\end{itemize}
\end{frame}
\begin{frame}[label={sec:orge196e78}]{Data Realities}
\begin{itemize}
\item Data always contains degrees of subjective bias
\item Data can \alert{approach} but never fully represent reality
\end{itemize}
\end{frame}
\begin{frame}[label={sec:org76ca171}]{The Rhetoricity of Data}
The collection, selection, and presentation is ultimately rhetorical, responding the needs of context-specific arguments and audiences.
\end{frame}
\begin{frame}[label={sec:orgf767ded}]{Data Visualization as Rhetoric}
\begin{itemize}
\item Visualizations both represent and construct the realities they depict.
\item Visualizations involve interpretive, persuasive choices.
\item Visualizations are arguments, involving audience consideration.
\end{itemize}
\end{frame}

\begin{frame}[label={sec:org769c88a}]{Data Tells Stories}
Numerical choices in displaying data ("counts, averages, ratios,or percentages") as well as social choices ("deciding whether to summarize data based on gender, race, political affiliation, or some other variable") are central to making rhetorical choices in data visualization (Wolfe, “Teaching Students to Focus on the Data in Data Visualization" 347).
\end{frame}
